# /ai-review 昇格ルール（/security-review を自動起動する条件）
#
# 形式: 1 行 1 ルール。「<種別> <正規表現>」。空行と # 行は無視。
#   path     変更ファイルのパス（git diff --name-only）に対する ERE。大文字小文字無視
#   content  追加行（git diff の + 行、先頭 + を除いた本文）に対する ERE。大文字小文字無視
#   secret   このパスが差分に含まれたら Codex（外部モデル）へ送らず警告する。大文字小文字無視
#   max_files   変更ファイル数の上限（超えたら昇格）
#   max_added   追加行数の上限（超えたら昇格）
#
# プロジェクト固有の追記は、対象リポジトリの .ai-review/escalation-rules.local.txt に同じ形式で書く。
# ここは全プロジェクト共通の既定。モデルは一切関与しない（grep のみ）。

# ---- A. パス ----
path  (^|/)(auth|authn|authz|login|logout|session|oauth|oidc|saml|sso|token|jwt|password|passwd|credential)s?(/|\.|-|_|$)
path  (^|/)(payment|billing|checkout|stripe|invoice|subscription|wallet|refund)s?(/|\.|-|_|$)
path  (^|/)(migration|migrate|schema|prisma|drizzle|sequelize|typeorm|alembic)s?(/|\.|-|_|$)
path  \.sql$
path  (^|/)(middleware|rbac|acl|permission|policy|guard|interceptor)s?(/|\.|-|_|$)
path  ^\.github/workflows/
path  (^|/)Dockerfile
path  (^|/)docker-compose.*\.ya?ml$
path  (^|/)\.env\.example$
path  (^|/)(package|composer)\.json$
path  (^|/)(package-lock\.json|pnpm-lock\.yaml|yarn\.lock|Cargo\.lock|poetry\.lock|go\.sum|Gemfile\.lock)$
path  (^|/)(requirements[^/]*\.txt|pyproject\.toml|go\.mod|Cargo\.toml|Gemfile)$

# ---- B. 追加行の内容 ----
content  \b(exec|execSync|spawn|spawnSync|execFile|system|popen|subprocess\.(run|call|Popen)|Runtime\.getRuntime\(\)\.exec|ProcessBuilder)\s*\(
content  \bchild_process\b
content  \beval\s*\(
content  \bnew\s+Function\s*\(
content  \b(crypto|cipher|decipher|createHash|createHmac|jwt|jsonwebtoken|bcrypt|argon2|scrypt|pbkdf2|md5|sha1|MessageDigest|Cipher\.getInstance|SecureRandom|Math\.random)\b
content  (SELECT|INSERT|UPDATE|DELETE|DROP|ALTER)\b[^;]*['"]\s*\+|['"]\s*\+\s*[^;]*\b(WHERE|FROM|VALUES)\b
content  \$\{[^}]*\}[^;]*\b(SELECT|INSERT|UPDATE|DELETE|WHERE)\b|\b(SELECT|INSERT|UPDATE|DELETE|WHERE)\b[^;]*\$\{
content  \b(fetch|axios|got|request|http\.get|https\.get|urllib|requests\.(get|post))\s*\(\s*[^'")]*\b(req|request|params|query|body|input|url|href)\b
content  \bfs\.(readFile|readFileSync|writeFile|writeFileSync|createReadStream|createWriteStream|unlink|rm|readdir)\s*\([^)]*\b(req|request|params|query|body|input|name|path)\b
content  \bpath\.(join|resolve)\s*\([^)]*\b(req|request|params|query|body|input)\b
content  \bSet-Cookie\b|\bres\.cookie\s*\(|\bcookies?\.set\s*\(
content  Access-Control-Allow-Origin|\bcors\s*\(
content  \bdangerouslySetInnerHTML\b|\binnerHTML\s*=|\bbypassSecurityTrust|\bv-html\b
content  \b(deserialize|unserialize|pickle\.loads?|yaml\.load|ObjectInputStream|readObject)\s*\(
content  \b(disableCsrf|csrf\s*:\s*false|verify\s*:\s*false|rejectUnauthorized\s*:\s*false|InsecureSkipVerify|CURLOPT_SSL_VERIFYPEER,\s*false)\b

# ---- 秘密情報（Codex へ送らない） ----
secret  (^|/)\.env(\..*)?$
secret  \.(pem|key|p12|pfx|crt|cer|cert|jks|keystore)$
secret  (^|/)(id_rsa|id_ed25519|id_ecdsa)(\.pub)?$
secret  (^|/)(secrets?|credentials?)\.(json|ya?ml|toml)$

# ---- C. 規模 ----
max_files  20
max_added  1000
